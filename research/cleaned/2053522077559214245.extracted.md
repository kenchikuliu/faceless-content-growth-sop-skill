### Prompt 6

```
"""
publish.py
Posts one video to YouTube, Instagram and TikTok in a single command.

Usage:
    python publish.py --video ./out/video.mp4 \
                      --short-video ./out/short.mp4 \
                      --short-public-url https://your.cdn/short.mp4 \
                      --metadata ./out/meta.json

Required env vars:
    YT_CLIENT_SECRETS    path to OAuth client secrets JSON from Google Cloud
    IG_USER_ID           Instagram Business account ID
    IG_ACCESS_TOKEN      long lived Graph API token
    TT_ACCESS_TOKEN      TikTok Content Posting API token
"""

import os, json, time, argparse, requests
from googleapiclient.discovery import build
from googleapiclient.http import MediaFileUpload
from google_auth_oauthlib.flow import InstalledAppFlow
from google.auth.transport.requests import Request
from google.oauth2.credentials import Credentials


def upload_youtube(video_path, meta):
    SCOPES = ["https://www.googleapis.com/auth/youtube.upload"]
    creds = None
    if os.path.exists("yt_token.json"):
        creds = Credentials.from_authorized_user_file("yt_token.json", SCOPES)
    if not creds or not creds.valid:
        if creds and creds.expired and creds.refresh_token:
            creds.refresh(Request())
        else:
            flow = InstalledAppFlow.from_client_secrets_file(
                os.environ["YT_CLIENT_SECRETS"], SCOPES)
            creds = flow.run_local_server(port=0)
        with open("yt_token.json", "w") as f:
            f.write(creds.to_json())

    yt = build("youtube", "v3", credentials=creds)
    body = {
        "snippet": {
            "title": meta["youtube"]["title"],
            "description": meta["youtube"]["description"],
            "tags": meta["youtube"]["tags"],
            "categoryId": meta["youtube"].get("category", "22"),
        },
        "status": {
            "privacyStatus": meta["youtube"].get("privacy", "public"),
            "selfDeclaredMadeForKids": False,
        },
    }
    media = MediaFileUpload(video_path, chunksize=-1, resumable=True)
    request = yt.videos().insert(part="snippet,status", body=body, media_body=media)
    response = None
    while response is None:
        status, response = request.next_chunk()
        if status:
            print(f"  YouTube {int(status.progress() * 100)}%")
    return f"https://youtu.be/{response['id']}"


def upload_instagram(public_video_url, meta):
    user_id = os.environ["IG_USER_ID"]
    token = os.environ["IG_ACCESS_TOKEN"]

    create = requests.post(
        f"https://graph.facebook.com/v19.0/{user_id}/media",
        data={
            "media_type": "REELS",
            "video_url": public_video_url,
            "caption": meta["instagram"]["caption"],
            "access_token": token,
        },
    ).json()
    container_id = create["id"]

    for _ in range(20):
        s = requests.get(
            f"https://graph.facebook.com/v19.0/{container_id}",
            params={"fields": "status_code", "access_token": token},
        ).json()
        if s.get("status_code") == "FINISHED":
            break
        time.sleep(15)

    publish = requests.post(
        f"https://graph.facebook.com/v19.0/{user_id}/media_publish",
        data={"creation_id": container_id, "access_token": token},
    ).json()
    return publish.get("id")


def upload_tiktok(video_path, meta):
    token = os.environ["TT_ACCESS_TOKEN"]
    file_size = os.path.getsize(video_path)

    init = requests.post(
        "https://open.tiktokapis.com/v2/post/publish/video/init/",
        headers={"Authorization": f"Bearer {token}"},
        json={
            "post_info": {
                "title": meta["tiktok"]["caption"],
                "privacy_level": "PUBLIC_TO_EVERYONE",
                "disable_duet": False,
                "disable_comment": False,
                "disable_stitch": False,
            },
            "source_info": {
                "source": "FILE_UPLOAD",
                "video_size": file_size,
                "chunk_size": file_size,
                "total_chunk_count": 1,
            },
        },
    ).json()
    upload_url = init["data"]["upload_url"]
    publish_id = init["data"]["publish_id"]

    with open(video_path, "rb") as f:
        requests.put(
            upload_url,
            headers={
                "Content-Range": f"bytes 0-{file_size - 1}/{file_size}",
                "Content-Type": "video/mp4",
            },
            data=f.read(),
        )
    return publish_id


def main():
    p = argparse.ArgumentParser()
    p.add_argument("--video", required=True, help="Long form mp4")
    p.add_argument("--short-video", required=True, help="Vertical 9:16 mp4")
    p.add_argument("--short-public-url", required=True, help="Public URL for IG")
    p.add_argument("--metadata", required=True, help="meta.json from Claude")
    args = p.parse_args()

    with open(args.metadata) as f:
        meta = json.load(f)

    results = {}
    for name, fn in [
        ("youtube", lambda: upload_youtube(args.video, meta)),
        ("instagram", lambda: upload_instagram(args.short_public_url, meta)),
        ("tiktok", lambda: upload_tiktok(args.short_video, meta)),
    ]:
        print(f">>> {name}")
        try:
            results[name] = fn()
            print(f"  done: {results[name]}")
        except Exception as e:
            results[name] = f"failed: {e}"
            print(f"  failed: {e}")

    print("\n=== Results ===")
    print(json.dumps(results, indent=2))


if __name__ == "__main__":
    main()
```
